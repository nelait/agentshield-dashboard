3. Infrastructure Setup (Step-by-Step)

1

Resource Group & Extensions

~3 minutes
# Install required CLI extensions
az extension add --name containerapp --upgrade
az extension add --name rdbms-connect --upgrade

# Set defaults
export RG=agentshield-poc
export LOCATION=eastus

# Create resource group
az group create --name $RG --location $LOCATION

# Register required providers
az provider register --namespace Microsoft.App
az provider register --namespace Microsoft.OperationalInsights
az provider register --namespace Microsoft.DBforPostgreSQL
az provider register --namespace Microsoft.Cache
2

Database & Cache

~10 minutes
# Generate a strong password
export DB_PASSWORD="$(openssl rand -base64 24)"

# Create Azure Database for PostgreSQL (Flexible Server)
az postgres flexible-server create \
  --resource-group $RG \
  --name agentshield-db \
  --location $LOCATION \
  --admin-user agentshield \
  --admin-password "$DB_PASSWORD" \
  --sku-name Standard_B1ms \
  --tier Burstable \
  --storage-size 32 \
  --version 16 \
  --public-access 0.0.0.0  # Allow Azure services; restrict later

# Create database
az postgres flexible-server db create \
  --resource-group $RG \
  --server-name agentshield-db \
  --database-name agentshield

# Create Azure Cache for Redis
az redis create \
  --resource-group $RG \
  --name agentshield-redis \
  --location $LOCATION \
  --sku Basic --vm-size c0 \
  --enable-non-ssl-port

# Get Redis access key
export REDIS_KEY=$(az redis list-keys --resource-group $RG \
  --name agentshield-redis --query primaryKey -o tsv)
3

Key Vault & Container Registry

~5 minutes
# Create Azure Key Vault for secrets
az keyvault create --resource-group $RG --name agentshield-kv --location $LOCATION

# Store secrets
az keyvault secret set --vault-name agentshield-kv \
  --name jwt-secret --value "$(openssl rand -hex 32)"

az keyvault secret set --vault-name agentshield-kv \
  --name db-password --value "$DB_PASSWORD"

az keyvault secret set --vault-name agentshield-kv \
  --name compliance-key --value "$(openssl rand -base64 32 | head -c 32)"

# Create Azure Container Registry
az acr create --resource-group $RG --name agentshieldacr \
  --sku Basic --admin-enabled true

# Build and push images directly from source (no local Docker needed!)
az acr build --registry agentshieldacr --image agentshield-api:latest \
  --file Dockerfile .
📋 Azure ACR Tasks — Cloud-Based Builds
az acr build uploads your source code and builds the Docker image in Azure — no local Docker daemon required. This is especially useful for teams on machines without Docker Desktop.
4

Deploy AgentShield Backend (Container Apps)

~10 minutes
# Create Container Apps Environment
az containerapp env create \
  --resource-group $RG \
  --name agentshield-env \
  --location $LOCATION

# Get DB & Redis connection info
export DB_HOST=agentshield-db.postgres.database.azure.com
export REDIS_HOST=agentshield-redis.redis.cache.windows.net

# Deploy AgentShield API as Container App
az containerapp create \
  --resource-group $RG \
  --name agentshield-api \
  --environment agentshield-env \
  --image agentshieldacr.azurecr.io/agentshield-api:latest \
  --registry-server agentshieldacr.azurecr.io \
  --target-port 3000 \
  --ingress external \
  --min-replicas 1 --max-replicas 4 \
  --cpu 1.0 --memory 2.0Gi \
  --env-vars \
    NODE_ENV=production \
    DB_HOST=$DB_HOST \
    DB_NAME=agentshield \
    DB_USER=agentshield \
    DB_SSL=true \
    DB_PORT=5432 \
    REDIS_HOST=$REDIS_HOST \
    REDIS_PORT=6380 \
    REDIS_PASSWORD=$REDIS_KEY \
    LOG_LEVEL=info \
  --secrets \
    jwt-secret="$(az keyvault secret show --vault-name agentshield-kv --name jwt-secret --query value -o tsv)" \
    db-password="$DB_PASSWORD" \
    compliance-key="$(az keyvault secret show --vault-name agentshield-kv --name compliance-key --query value -o tsv)" \
  --secret-env-vars \
    JWT_SECRET=jwt-secret \
    DB_PASSWORD=db-password \
    COMPLIANCE_ENCRYPTION_KEY=compliance-key \
  --scale-rule-name http-scaling \
  --scale-rule-type http \
  --scale-rule-http-concurrency 50

# Get the app URL
export API_URL=$(az containerapp show --resource-group $RG \
  --name agentshield-api --query properties.configuration.ingress.fqdn -o tsv)
echo "API URL: https://$API_URL"

Run Migrations (One-time Job)

# Create a Container Apps Job for migrations
az containerapp job create \
  --resource-group $RG \
  --name migrate-db \
  --environment agentshield-env \
  --image agentshieldacr.azurecr.io/agentshield-api:latest \
  --registry-server agentshieldacr.azurecr.io \
  --trigger-type Manual \
  --cpu 0.5 --memory 1.0Gi \
  --env-vars DB_HOST=$DB_HOST DB_NAME=agentshield DB_USER=agentshield DB_SSL=true \
  --secret-env-vars DB_PASSWORD=db-password \
  --secrets db-password="$DB_PASSWORD" \
  --args "npm" "run" "migrate"

# Execute the migration job
az containerapp job start --resource-group $RG --name migrate-db

# Seed admin account (same pattern)
az containerapp job create --name seed-db ... --args "npm" "run" "seed"
az containerapp job start --resource-group $RG --name seed-db

Deploy Test Agents (Container Apps)

~15 minutes
# Build each agent image in ACR
for agent in echo-agent finance-agent support-agent code-review-mcp weather-mcp; do
  az acr build --registry agentshieldacr \
    --image agentshield-$agent:latest \
    --file ./agents/$agent/Dockerfile ./agents/$agent
done

# Deploy each agent as a Container App (scale-to-zero enabled)
for agent in echo-agent finance-agent support-agent code-review-mcp weather-mcp; do
  az containerapp create \
    --resource-group $RG \
    --name $agent \
    --environment agentshield-env \
    --image agentshieldacr.azurecr.io/agentshield-$agent:latest \
    --registry-server agentshieldacr.azurecr.io \
    --target-port 8080 \
    --ingress internal \
    --min-replicas 0 --max-replicas 2 \
    --cpu 0.25 --memory 0.5Gi
done

# Agents are reachable within the Container Apps Environment at:
# https://echo-agent.internal.agentshield-env.azurecontainerapps.io
# https://finance-agent.internal.agentshield-env.azurecontainerapps.io
# etc.
📋 Container Apps Internal Ingress
Using --ingress internal makes test agents accessible only within the Container Apps Environment — not from the public internet. AgentShield reaches them via internal DNS names. Scale-to-zero means agents only run (and cost money) when invoked.

4. Test Agent Design

We create 6 purpose-built test agents, each designed to trigger specific AgentShield governance features. Together they cover REST, MCP, and external (OpenAI-compatible) protocols.

🔄 Echo Agent
Echoes back any input. Baseline agent for policy, auth, and audit testing.
Protocol: RESTSlug: echo-agentPurpose: Policy enforcement, auth, audit logging
💰 Finance Agent
Returns mock financial data with configurable token usage in responses.
Protocol: REST (with usage object)Slug: finance-agentPurpose: Cost tracking, budget enforcement
📞 Support Agent
Simulates customer support — intentionally returns PII and sensitive data.
Protocol: RESTSlug: support-agentPurpose: Guardrails (PII, profanity, injection)
💻 Code Review MCP
MCP server that exposes code review tools (analyze, lint, security-scan).
Protocol: MCP (SSE transport)Slug: code-review-mcpTools: analyze_code, lint, security_scan
🌤️ Weather MCP
MCP server providing weather data tools. Lightweight, good for health checks.
Protocol: MCP (SSE transport)Slug: weather-mcpTools: get_weather, get_forecast
🧠 OpenAI Proxy
Proxies to a real OpenAI-compatible endpoint. Returns real usage data.
Protocol: REST (OpenAI-compatible)Slug: openai-proxyAuth: Bearer token (API key)
💡 Agent Source Code
The test agent implementations are identical across AWS, Azure, and GCP — they are standard Node.js Express apps packaged in Docker containers. See the GCP POC Guide § 5 for full source code of each agent.

6. Agent Registration & Configuration

# Set base URL (Container Apps FQDN)
export API="https://agentshield-api.agentshield-env.azurecontainerapps.io/api/v1"

# Login → get JWT token
TOKEN=$(curl -s -X POST $API/auth/login \
  -H "Content-Type: application/json" \
  -d '{"email":"admin@agentshield.local","password":"admin123"}' \
  | jq -r .data.accessToken)

# Register each agent (using Container Apps internal DNS names)
curl -X POST $API/admin/agents -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"name":"Echo Agent","slug":"echo-agent","type":"internal","protocol":"rest",
      "endpointUrl":"https://echo-agent.internal.agentshield-env.azurecontainerapps.io",
      "description":"Baseline echo agent for policy/auth testing"}'

curl -X POST $API/admin/agents -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"name":"Finance Agent","slug":"finance-agent","type":"internal","protocol":"rest",
      "endpointUrl":"https://finance-agent.internal.agentshield-env.azurecontainerapps.io",
      "description":"Financial analysis with token usage reporting"}'

curl -X POST $API/admin/agents -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"name":"Code Review MCP","slug":"code-review-mcp","type":"internal","protocol":"mcp",
      "endpointUrl":"https://code-review-mcp.internal.agentshield-env.azurecontainerapps.io/mcp/sse",
      "description":"Code review tools via MCP protocol"}'

7. Demo 1: Agent Registry Phase 1

Scenario 1.1 — Register and discover agents
Prove: Unified registry supports REST, MCP, and external protocols
  1. Register all 6 agents via dashboard or API
  2. Open Dashboard → Agent Registry — verify all 6 appear with correct protocol badges
  3. Click each agent — verify details (endpoint, protocol, description) are visible
✅ All agents visible with correct metadata. Protocol icons distinguish REST vs MCP.
Scenario 1.2 — Health monitoring
Prove: AgentShield actively monitors agent availability
  1. Wait 30 seconds for health checks to run
  2. Verify agents show "healthy" status
  3. Scale an agent to 0 replicas: az containerapp update --min-replicas 0 --max-replicas 0
  4. Wait for 3 consecutive failures — "unhealthy" (red)
  5. Scale back — status returns to "healthy"
✅ Health status transitions: healthy → unhealthy → healthy.
Scenario 1.3 — Agent invocation through gateway
Prove: Requests are proxied transparently through AgentShield
  1. Invoke Echo Agent: POST /api/v1/gateway/agents/echo-agent/invoke
  2. Verify response contains X-Forwarded-By: AgentShield
  3. Invoke Code Review MCP — verify MCP tool listing and tool call works
✅ Transparent proxying works for both REST and MCP protocols.
